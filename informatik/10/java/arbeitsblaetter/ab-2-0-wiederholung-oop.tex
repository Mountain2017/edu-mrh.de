\begin{java}
void geheNachRechts() {
   setAngle(0);
   move(10, 0);
}

void geheNachOben() {
   setAngle(270);
   move(0, -10);
}
\end{java}

\begin{minipage}[t]{0.48\linewidth}
\begin{java}
// ca. 30-mal pro Sekunde
void act() {
   if (isKeyDown("w")) {
      geheNachOben();
   } else if (isKeyDown("d")) {
      geheNachRechts();
   }
   esseKorn();
}
\end{java}
\end{minipage}
\lsglinien{3}{Solange w gedrückt ist, dreht sich der Hamster nach oben und läuft 10 Pixel nach oben, sonst bei d nach rechts (w hat Vorrang). Danach frisst er ein Korn, falls er eins berührt. Das passiert ca. 30-mal pro Sekunde.}
