\begin{java}
class Zeichner extends Turtle {

   Zeichner(int x, int y) {
      super(x, y, true);
      setBorderColor(Color.aqua);
      setBorderWidth(5);
      penDown();
   }

   void quadrat(double seitenlaenge) {
      for (int i = 0; i < 4; i++) {
         forward(seitenlaenge);
         turn(90); // gegen den Uhrzeigersinn
      }
   }
}
\end{java}

\textbf{Wiederholung mit fester Anzahl $\leftrightarrow$ bedingte Wiederholung}

\begin{minipage}[t]{0.48\linewidth}
\textbf{Wiederholung mit fester Anzahl}
\begin{java}
for (int i = 0; i < 3; i = i + 1)
{
   erstelleQuadratAn(i, 1);
}
\end{java}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
\textbf{Wiederholung mit Bedingung}
\begin{java}
int i = 0;
while (i < 3)
{
   erstelleQuadratAn(i, 1);
   i = i + 1;
}
\end{java}
\end{minipage}
