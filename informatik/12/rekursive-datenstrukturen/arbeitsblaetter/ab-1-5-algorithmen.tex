\abtitel{1.5}{Rekursive Algorithmen}

Ruft sich eine Methode selbst mit anderem Parameter auf, ist sie \textbf{rekursiv}. \web{Station 1.5, Labor und Projekt}

\begin{aufgabe}[Hin und zurück]{1}
\begin{minipage}[t]{0.43\linewidth}
\vspace{0pt}
\begin{java}[basicstyle=\ttfamily\footnotesize]
void countdown(int n) {
   if (n == 0) {
      println("Start!");
   } else {
      println("hin " + n);
      countdown(n - 1);
      println("zurück " + n);
   }
}
\end{java}
\end{minipage}\hfill
\begin{minipage}[t]{0.54\linewidth}
\vspace{0pt}\lueckentext
\textbf{a)} Sage vorher, was \code{countdown(3)} ausgibt:\par
\lsgrest{hin 3, hin 2, hin 1, Start!,}
\lsgrest{zurück 1, zurück 2, zurück 3}
\textbf{b)} Wenn „Start!“ erscheint, warten \lsg[2.25cm]{3} Aufrufe. Als Erstes gibt \lsg[4.6cm]{countdown(1)} „zurück“ aus.
\end{minipage}
\end{aufgabe}

\begin{merke}[Rekursive Algorithmen]
\textbf{Drei Fragen:} \nr{1} \textbf{Abbruchbedingung}: Wann steht die Antwort sofort fest? \nr{2} \textbf{Verkleinern}: Mit welchem kleineren Problem ruft sich die Methode auf? \nr{3} \textbf{Zusammensetzen}: Was macht sie mit der Antwort?\par
\lueckentext Wartende Aufrufe liegen auf dem \lsg[5.25cm]{Aufrufstapel}, der zuletzt aufgerufene wird als \lsg[2.25cm]{Erster} fertig.\par
\textbf{Linear:} höchstens \lsg[2.25cm]{ein} rekursiver Aufruf je Durchlauf, Darstellung: \lsg[5.25cm]{Aufruftabelle}.
\textbf{Verzweigt:} \lsg[2.25cm]{mehrere} rekursive Aufrufe je Durchlauf, Darstellung: \lsg[5.25cm]{Aufrufbaum}.
\end{merke}

\begin{aufgabe}[Wenn die Rekursion nicht endet (Projekt, Klasse Experiment)]{2}
\textbf{a)} Starte jede Fassung mit \code{n = 3} und notiere die ersten fünf Werte von \code{n}. \textbf{b)} Welche der drei Fragen \nr{1}–\nr{3} ist verletzt? Begründe kurz.\par\vspace{1mm}
{\small\renewcommand{\arraystretch}{1.1}
\begin{tabularx}{\linewidth}{|l|>{\ttfamily\footnotesize}l|p{3.2cm}|X|}\hline
 & \textrm{\small Kern des Codes} & a) erste Werte von \code{n} & b) verletzt, weil … \\\hline
A & (kein if) fassungA(n - 1) & \schreibzeile\lsgtext{3, 2, 1, 0, -1} & \lsgtext{\nr{1}: Es gibt keine Abbruchbedingung.} \\\hline

\end{tabularx}}\par\vspace{1mm}

\end{lueckentext}
\end{aufgabe}

\begin{aufgabe}[Fibonacci: verzweigte Rekursion (TODO 5a, 5b)]{5}

\vspace{0pt}
\textbf{a)} Ergänze den Code.
\begin{javaluecke}[basicstyle=\linespread{1.5}\footnotesize\ttfamily]
int fib(int n) {
   aufrufe++;                   // TODO 5b
   if (§\lsg[2.25cm]{n == 0}§) { return 0; }
   if (n == 1) { return §\lsg[1.2cm]{1}§; }
   return §\lsg[4.5cm]{fib(n - 1) + fib(n - 2)}§;
