\abtitel{4}{Das Perzeptron: eine Grenze im Raum}

\begin{aufgabe}[Nettoeingabe und Ausgabe]{1}
Ein Perzeptron hat die Gewichte $w_0 = -0{,}5$, $w_1 = -1$, $w_2 = 1{,}5$. Es gilt $z = w_0\cdot 1 + w_1 x_1 + w_2 x_2$ und Ausgabe 1, wenn $z \ge 0$, sonst 0.\par\vspace{1mm}
\renewcommand{\arraystretch}{1.1}
\begin{tabularx}{\linewidth}{|c|c|X|c|}\hline
$x_1$ & $x_2$ & Nettoeingabe $z$ (mit Rechnung) & Ausgabe \\\hline
0,2 & 0,8\schreibzeile & \lsgklein{$-0{,}5 - 0{,}2 + 1{,}2 = 0{,}5$} & \lsgklein{1} \\\hline
0,7 & 0,3\schreibzeile & \lsgklein{$-0{,}5 - 0{,}7 + 0{,}45 = -0{,}75$} & \lsgklein{0} \\\hline
0,4 & 0,6\schreibzeile & \lsgklein{$-0{,}5 - 0{,}4 + 0{,}9 = 0$ (genau auf der Grenze)} & \lsgklein{1} \\\hline
0,5 & 0,5\schreibzeile & \lsgklein{$-0{,}5 - 0{,}5 + 0{,}75 = -0{,}25$} & \lsgklein{0} \\\hline
\end{tabularx}
\end{aufgabe}

\begin{aufgabe}[Merkkasten]{3}
\begin{lueckentext}
Die Nettoeingabe $z$ wird an eine \lsg[5.25cm]{Aktivierungsfunktion} weitergegeben, beim Perzeptron die \lsg[5.25cm]{Stufenfunktion}. Die Punkte mit $z = 0$ bilden die \lsg[5.25cm]{Entscheidungsgrenze}: bei zwei Merkmalen eine \lsg[3.6cm]{Gerade}, bei drei eine \lsg[3.6cm]{Ebene}. Ein Perzeptron kann daher nur \lsg[4.5cm]{linear} trennen. Der Gewichtsvektor steht \lsg[4.5cm]{senkrecht} auf der Grenze, der Bias $w_0$ \lsgrest{verschiebt die Grenze.}
\end{lueckentext}
\end{aufgabe}

\begin{aufgabe}[Ein Lernschritt von Hand]{4}
Gewichte $(w_0 \,|\, w_1 \,|\, w_2) = (0{,}5 \,|\, 1 \,|\, -1)$, Lernrate $\eta = 0{,}1$. Der Punkt $x = (0{,}6 \,|\, 0{,}2)$ hat das Ziel 0. Lernregel: $\Delta w_i = \eta \cdot (\text{Ziel} - \text{Ausgabe}) \cdot x_i$ mit $x_0 = 1$.
\begin{lueckentext}
$z = $ \lsg[4.5cm]{$0{,}5 + 0{,}6 - 0{,}2 = 0{,}9$}, Ausgabe \lsg[2.25cm]{1}, Ziel $-$ Ausgabe $=$ \lsg[2.25cm]{$-1$}\\
$\Delta w_0 = $ \lsg[3cm]{$-0{,}1$}, \quad $\Delta w_1 = $ \lsg[3cm]{$-0{,}06$}, \quad $\Delta w_2 = $ \lsg[3cm]{$-0{,}02$}\\
neue Gewichte: $(w_0 \,|\, w_1 \,|\, w_2) = $ \lsg[5.25cm]{$(0{,}4 \,|\, 0{,}94 \,|\, -1{,}02)$}
\end{lueckentext}
