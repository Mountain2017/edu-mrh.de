% ------------------------------------------------------------------ 0
\begin{wsbox}{0}{Know the task — the operator tells you what to do}
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{30mm}>{\raggedright\arraybackslash}p{45mm}>{\raggedright\arraybackslash}X@{}}
\textbf{I Comprehension} & describe, outline, sum up & \emph{What} does the text say?\\
\textbf{II Analysis} & analyse, examine, explain & \emph{How} does it say it — and why?\\
\textbf{III Evaluation} & comment, discuss, assess & What do \emph{you} think — and why?\\
\end{tabularx}
\end{wsbox}

% ------------------------------------------------------------------ 1
\begin{wsbox}{1}{Step 1 · Preparation}
\begin{minipage}[t]{0.40\linewidth}\vspace{0pt}
\begin{enumerate}[label=\textcolor{prim}{\textbf{\alph*}},leftmargin=1.2em,itemsep=1mm]
\item Read the \textbf{questions} first, mark the operators.
\item \textbf{Skim}: headline, lead, first and last paragraph.
\item \textbf{Read} the whole text, mark key words.
\item \textbf{Purpose}: text type, source, date, target group.
\item \textbf{Understand}: 3–6 words per section in the margin.
\end{enumerate}
\end{minipage}\hfill

% ------------------------------------------------------------------ 2
\begin{wsbox}{2}{Step 2 · Writing}
\kl{2a\enspace Introductory sentence} {\footnotesize (first answer only)}\par\vspace{0.8mm}
\begin{tintbox}
\small In the \textbf{[text type]} “\textbf{[title]}”, published in \textbf{[source]} in \textbf{[date]}, \textbf{[author]} \textbf{[reporting verb]} \textbf{[topic]}.
\end{tintbox}
\lsglinien{2}{In her feature article “Helsinki's huge VR gig and the potential of virtual tourism”, published in The Guardian online in 2020, Sarah Baxter reports on a virtual concert and explores the future of virtual tourism.}

\vspace{2mm}
\begin{minipage}[t]{0.47\linewidth}\vspace{0pt}
\kl{2b\enspace Comprehension (I)}\par\vspace{0.6mm}
\begin{itemize}[itemsep=0.6mm]
\item only what is asked
\item own words + line references
\item present tense, no opinion
\end{itemize}
\vspace{1mm}
{\small Our extra rule:}
\lsglinien{1}{Pick up the question in the first sentence.}
\end{minipage}\hfill
\begin{minipage}[t]{0.5\linewidth}\vspace{0pt}
\kl{2c\enspace Analysis (II)}\par\vspace{1.2mm}
\begin{tikzpicture}[x=1mm,y=1mm,font=\footnotesize,
  stufe/.style={rounded corners=1mm,fill=primtint,minimum height=10mm,text width=24mm,align=center,inner sep=1pt}]
  \node[stufe] (c) at (0,0) {\textbf{Claim}\\[-0.4mm]{\scriptsize what the author does}};

{\small Thesis \pfeil one paragraph per aspect \pfeil conclusion}
\end{minipage}

\vspace{2.5mm}
\textcolor{sec}{\small\textbf{Our model paragraph} (Baxter: respect for Helsinki's achievements) — mark C, E, E:}
\lsglinien{5}{(C) Baxter expresses respect for Helsinki's achievements through appreciative vocabulary. (E) Although she admits that the gig lacked “the buzz of being at a ‘proper’ concert” (ll.\,11–12), she stresses that “technically, it was impressive” (ll.\,12–13). (E) Because she first concedes a weakness, her praise sounds balanced and credible, so readers are more likely to share her positive view.}
\end{wsbox}

% ------------------------------------------------------------------ Step 3
\begin{wsbox}{4}{Step 3 · Editing and checking}
\begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
\begin{itemize}[label=$\square$,itemsep=1mm]
\item I did exactly what the operator asks.
\item Every statement has a line reference.
\item Quotes are exact and marked “…”.
\item My opinion only where asked (III).
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
\begin{itemize}[label=$\square$,itemsep=1mm]
\item present tense, no contractions
\item linking words show my line of thought
\item precise verbs instead of \emph{says}
\item no repetition
\end{itemize}
\end{minipage}
\end{wsbox}
