% ------------------------------------------------------------------ 2
\begin{wsbox}{2}{Artificial intelligence (AI)}
\begin{tintbox}
AI = \emph{the simulation of human intelligence in machines}: learning, problem-solving, \emph{reasoning}, perceiving.
\end{tintbox}\par\vspace{1mm}
\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X@{\hspace{4mm}}>{\raggedright\arraybackslash}X@{\hspace{4mm}}>{\raggedright\arraybackslash}X@{}}
\kl{Weak / narrow AI} & \kl{Strong / general AI} & \kl{Impact}\\
single tasks, e.g.\ personal assistants, video games &
complex, human-like, e.g.\ (book) self-driving cars, robots\textsuperscript{*} &
\plus everyday life, education, health\newline \minus jobs, deepfakes, misinformation, copyright
\end{tabularx}\par\vspace{0.6mm}
{\footnotesize\textsuperscript{*}Most experts today call self-driving cars \emph{narrow} AI; \emph{general AI} means a still hypothetical human-level intelligence.}
\par\vspace{1mm}
\textcolor{sec}{\small\textbf{Connect:} Why does checking data matter more than ever now that AI creates content?}
\lsglinien{2}{AI can produce convincing texts and deepfakes, and it is often impossible to tell where content comes from. Checking sources is the only protection against manipulation and misinformation.}
\end{wsbox}

\begin{wsbox}{3}{Augmented reality (AR) and virtual reality (VR) — both extended reality (XR)}
\renewcommand{\arraystretch}{1.2}
\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{20mm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
 & \kl{AR} & \kl{VR}\\\arrayrulecolor{rule}\midrule
What it does & adds digital \emph{overlays} to the real world & \emph{replaces} the real world with a 360° environment\\
Devices & phone, tablet (camera) & headset, motion controllers\\
Used in & entertainment, medicine, shopping & gaming, tourism, education, medicine, \emph{warfare}\\
\end{tabularx}\par\vspace{1mm}
\textcolor{sec}{\small\textbf{Connect:} How are AR, VR and AI related? Name one example you know (e.g.\ Virtual Helsinki, p.\,230).}
\lsglinien{2}{AR and VR are both types of XR; AI can make them smarter and more realistic (e.g.\ AI avatars, generated worlds). Example: Virtual Helsinki is VR, a digital twin of the city.}
\end{wsbox}

% ------------------------------------------------------------------ 4
\begin{wsbox}{4}{Utopia or dystopia — where will technology take us?}
\begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
\begin{goodbox}
\kl{\textcolor{good}{Utopia}} \enspace \emph{ou-topos} = “no place”, \emph{eu-topos} = “good place” (Thomas More, 1516).\par
An \textbf{ideal society} that criticises the present by showing a better world.\par
{\footnotesize e.g.\ Butler, \emph{Erewhon} (1872); Bellamy, \emph{Looking Backward} (1888)}
\end{goodbox}
\end{minipage}\hfill
\begin{minipage}[t]{0.485\linewidth}\vspace{0pt}
\begin{badbox}
\kl{\textcolor{bad}{Dystopia}} \enspace \emph{dys-topos} = “bad place”.\par
A \textbf{worst-case society}, often in the near future, that warns about \emph{contemporary} problems.\par
{\footnotesize e.g.\ \emph{Brave New World} (1932), \emph{Nineteen Eighty-Four} (1949), \emph{The Hunger Games} (2008)}
\end{badbox}
\end{minipage}
\par\vspace{1.2mm}
\textcolor{sec}{\small\textbf{Connect:} Which technology from boxes 1–3 could be the starting point of a dystopia? Describe that society in one or two sentences.}
\lsglinien{3}{E.g.\ AI + VR: people spend most of their lives in virtual worlds whose content is selected by AI. Reality loses importance and opinions can be steered without anyone noticing — a society like \emph{Brave New World}, kept happy and passive by technology.}
\end{wsbox}
