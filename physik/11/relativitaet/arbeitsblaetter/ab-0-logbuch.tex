\textbf{\nr{2} Das Eigen-System finden}\hfill\weblink{Station 4 und 5}
\lsglinien{3}{Zeit: Für wen passieren beide Ereignisse am selben Ort? Der misst die Eigenzeit $\Delta t_0$. Länge: Wer ruht relativ zum Gegenstand? Der misst die Eigenlänge $L_0$.}

\vspace{1mm}
\textbf{\nr{3} Mit $\gamma$ umrechnen}\hfill\weblink{Station 4 und 5}
\lsglinien{2}{Alle anderen messen: Zeit mal $\gamma$: $\Delta t = \gamma\cdot\Delta t_0$. Länge durch $\gamma$: $L = L_0/\gamma$ (nur in Bewegungsrichtung).}
\end{planbox}
