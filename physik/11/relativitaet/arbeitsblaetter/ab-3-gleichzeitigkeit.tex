\abtitel{3}{Was heißt „gleichzeitig“?}
\leitfrage{Wie stellt man fest, dass zwei Dinge an verschiedenen Orten zur selben Zeit passieren?}

\begin{aufgabe}[Uhren stellen]{1}
\lueckentext
\textbf{a)} Funkuhr, 300 km vom Sender: Laufzeit $t = $ \lsg[5.5cm]{300 km : 300\,000 km/s = 0,001 s}. Die Uhr geht 1 ms \lsg[2.25cm]{nach}.\\
\textbf{b)} Zum Korrigieren braucht man \lsg[5.25cm]{die Entfernung} und Postulat \lsg[1cm]{2}: Das Signal läuft immer mit \lsg[1cm]{c}.\\
\textbf{c)} Zwei entfernte Uhren stellt man gleich, indem man \lsgrest{einen Lichtblitz genau aus der Mitte schickt:}
\lsgrest{Er ist zu beiden gleich lang unterwegs, beide Uhren starten bei seiner Ankunft.}
\end{aufgabe}

\begin{merke}[Ereignis \quad·\quad Messen ist nicht sehen]
\lueckentext
Ein \textbf{Ereignis} hat einen \lsg[2.5cm]{Ort} und einen \lsg[3.2cm]{Zeitpunkt} (Was? Wo? Wann?), gemessen in einem Bezugssystem.\\
Die Laufzeit des Lichts zum Beobachter wird immer \lsg[4.5cm]{herausgerechnet}. Was dann noch übrig bleibt, ist keine \lsg[5.25cm]{optische Täuschung}.
\end{merke}

\lueckentext
\textbf{b)} Für Bea läuft das Licht nach hinten mit \lsg[1.5cm]{c} und nach vorn mit \lsg[1.5cm]{c} (Postulat \lsg[1cm]{2}). Im Vergleich zu den Bällen fehlt:\\ \lsgrest{der Ausgleich durch $u = u' + v$. Die hintere Tür kommt dem Licht entgegen.}
\textbf{c)} $v = 0{,}5\,c$, halbe Wagenlänge $L$ (von Bea gemessen): Der Abstand zur hinteren Tür schrumpft mit \lsg[2.5cm]{$c + v = 1{,}5\,c$}, zur vorderen mit \lsg[2.5cm]{$c - v = 0{,}5\,c$}.\\
\phantom{\textbf{c)}} $t_\text{hinten} = $ \lsg[3cm]{$L/(1{,}5\,c)$} \quad $t_\text{vorn} = $ \lsg[3cm]{$L/(0{,}5\,c)$} \quad Verhältnis \lsg[2.25cm]{1 : 3}\\
\textbf{d)} Im Alltag fällt das nicht auf, weil \lsgrest{für $v \ll c$ gilt: $c + v \approx c - v$.}
Wer hat recht, Bea oder Zoe? \lsgrest{Beide: Nach Postulat 1 ist kein System bevorzugt.}
\end{aufgabe}

\begin{merke}[Relativität der Gleichzeitigkeit]
\lueckentext
Zwei Uhren sind \textbf{synchron}, wenn ein Lichtsignal aus der \lsg[2.5cm]{Mitte} bei beiden zur gleichen Uhrzeit ankommt.\\
Zwei Ereignisse an \lsg[5.25cm]{verschiedenen Orten}, die in einem Inertialsystem gleichzeitig sind, sind in einem dazu bewegten Inertialsystem \lsg[5.25cm]{nicht gleichzeitig}.\\
Passieren zwei Ereignisse am \lsg[3.5cm]{selben Ort} gleichzeitig, sind sie es in jedem System.
\end{merke}

\begin{aufgabe}[Die Argumentkette in eigenen Worten \normalfont(erst auf der Website sortieren)]{4}
\lsglinien{4}{Für Zoe ruht der Wagen, das Licht läuft mit c über gleich lange Wege: gleichzeitig. Auch für Bea läuft es in beide Richtungen mit c (P2), aber die hintere Tür fährt ihm entgegen, die vordere läuft davon: hinten zuerst. Kein System ist bevorzugt (P1): Gleichzeitigkeit hängt vom Bezugssystem ab.}
\end{aufgabe}

\begin{aufgabe}[Umgekehrt]{5}
\bea{Bea} löst links und rechts von sich (gleich weit entfernt) zwei Blitzlichter aus, für sie gleichzeitig. \zoe{Zoe} fährt nach rechts vorbei. Für Zoe blitzt zuerst das \quad \kreuz{0} linke \quad \kreuz{1} rechte Blitzlicht, denn
\lsglinien{2}{in Zoes System fährt Bea nach links, dem linken Signal entgegen. Damit beide Signale Bea trotzdem im selben Moment erreichen, muss das rechte früher gestartet sein.}
\end{aufgabe}
