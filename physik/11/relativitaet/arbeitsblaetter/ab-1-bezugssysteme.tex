\abtitel{1}{Wer bewegt sich?}
\leitfrage{Kann man in einem gleichmäßig fahrenden Zug feststellen, dass man fährt?}

\linien{1}

\begin{merke}[Bezugssystem, Inertialsystem, Relativitätsprinzip]
\lueckentext
Jede Angabe von Ort, Zeit und Geschwindigkeit gilt relativ zu einem \lsg[5.25cm]{Bezugssystem}.\\
In einem \textbf{Inertialsystem} bleibt ein kräftefreier Körper in Ruhe oder bewegt sich\\ \lsgrest{geradlinig gleichförmig.}
Kein Inertialsystem ist ein System, das \lsgrest{beschleunigt (anfährt, bremst, eine Kurve fährt).}
\textbf{Relativitätsprinzip:} In allen Inertialsystemen laufen mechanische Vorgänge \lsg[2.6cm]{gleich} ab. Kein\\
mechanisches Experiment entscheidet, \lsgrest{welches System „wirklich ruht“.}
\textbf{Geschwindigkeitsaddition:} Bewegt sich S$'$ mit $v$ gegenüber S, gilt \lsg[3.5cm]{$u = u' + v$}.
\end{merke}

\begin{multicols}{2}

\begin{aufgabe}[Laufband am Flughafen]{4}
Das Laufband bewegt sich mit 1,5 m/s. \bea{Bea} steht daneben, \zoe{Zoe} geht auf dem Band mit 1,2 m/s.\\[1mm]
\lueckentext
\textbf{a)} System S: \lsg[4cm]{Halle (Bea)} \quad System S$'$: \lsg[4cm]{Laufband (Zoe)} \quad $v = $ \lsg[2.25cm]{1,5 m/s}\\
\textbf{b)} Zoe in Laufrichtung: $u = $ \lsgrest{1,2 m/s + 1,5 m/s = 2,7 m/s}
\phantom{\textbf{b)}} Zoe gegen die Laufrichtung: $u = $ \lsgrest{$-1{,}2$ m/s + 1,5 m/s = 0,3 m/s}
\textbf{c)} Zoe bleibt für Bea auf der Stelle, wenn $u' = $ \lsg[2.25cm]{$-1{,}5$ m/s}.
\end{aufgabe}
