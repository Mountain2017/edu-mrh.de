\leitfrage{Wo steckt die Relativitätstheorie schon heute in Technik, die du benutzt?}

\begin{aufgabe}[Wie dein Navi rechnet: Ortung ist Zeitmessung]{1}
\lueckentext
\textbf{a)} Satellit in 20\,200 km Höhe genau über dir: Laufzeit $t = $ \lsgrest{20\,200 km : 300\,000 km/s = 0,067 s}
\phantom{\textbf{a)}} Entfernung $=$ \lsg[4cm]{$c\cdot$ Laufzeit}. Das klappt nur wegen Postulat \lsg[1cm]{2}: \lsgrest{c ist für alle gleich.}
\textbf{b)} Uhrenfehler 1 ns $\to$ Entfernungsfehler \lsgrest{$3\cdot10^8$ m/s $\cdot\,10^{-9}$ s $= 0{,}3$ m}
\textbf{c)} Für Fahrspur-Genauigkeit (3 m) müssen die Uhren auf etwa \lsg[2.25cm]{10 ns} genau sein.
\end{aufgabe}

\begin{aufgabe}[Die Satellitenuhr: \bea{Handy = Bea}, \zoe{Satellit = Zoe}, $v = 3870$ m/s]{2}
Mein Tipp: Ohne Relativitäts-Korrektur läge das Navi nach einem Tag daneben um\\
\kreuz{0} Millimeter \quad \kreuz{0} etwa 1 m \quad \kreuz{0} etwa 100 m \quad \kreuz{1} mehr als 10 km\\[1mm]
\begin{hinweis}
Dein Taschenrechner zeigt $\gamma = 1$? Für $v \ll c$ gilt: \quad $\gamma - 1 \approx \dfrac{v^2}{2\,c^2}$. \quad Die Uhr geht in der Zeit $t$ um $(\gamma - 1)\cdot t$ nach.
\end{hinweis}
\lueckentext
\textbf{a)} $\gamma - 1 \approx$ \lsgrest{$3870^2 : \left(2\cdot(3{,}00\cdot10^8)^2\right) = 8{,}3\cdot10^{-11}$}
\phantom{\textbf{a)}} In einem Tag (86\,400 s) geht die Satellitenuhr nach um \lsgrest{$8{,}3\cdot10^{-11}\cdot 86\,400$ s $= 7{,}2$ µs}
\textbf{b)} Schwächere Schwerkraft in der Höhe: Uhr geht um 45,7 µs pro Tag \textit{schneller}. Zusammen:\\ \lsgrest{$+45{,}7$ µs $- 7{,}2$ µs $= +38{,}5$ µs pro Tag (die Satellitenuhr geht vor)}
\textbf{c)} Ortsfehler nach einem Tag: \lsgrest{$3{,}00\cdot10^8$ m/s $\cdot\,38{,}5\cdot10^{-6}$ s $\approx 11{,}5$ km}
\end{aufgabe}

\begin{aufgabe}[Grenzfall Alltag: War die Mechanik aus Klasse 10 falsch?]{4}
\textbf{a)} Setze eine Alltagsgeschwindigkeit ein ($v \ll c$, also $\gamma \approx 1$). Was wird aus den Formeln?\\[1mm]
{\renewcommand{\arraystretch}{1}\small
\begin{tabularx}{\linewidth}{|>{\centering\arraybackslash}X|>{\centering\arraybackslash}X|>{\centering\arraybackslash}X|}\hline
$\Delta t = \gamma\cdot\Delta t_0$ & $L = L_0/\gamma$ & $u = \dfrac{u' + v}{1 + u'v/c^2}$\rule[-4mm]{0pt}{10mm}\\\hline
\lsgtext{$\Delta t = \Delta t_0$}\schreibzeile & \lsgtext{$L = L_0$} & \lsgtext{$u = u' + v$}\\\hline
\end{tabularx}}

\vspace{1mm}
\textbf{b)} Meine Faustregel: Ab wann muss man relativistisch rechnen? Warum braucht GPS es schon bei 3,87 km/s?
\lsglinien{3}{Bis etwa 10\,\% von c weicht $\gamma$ um weniger als 1\,\% ab. Entscheidend ist aber die geforderte Genauigkeit: GPS braucht Nanosekunden, da zählt schon $\gamma - 1 \approx 10^{-10}$.}
\end{aufgabe}

\begin{merke}[Die klassische Mechanik als Grenzfall]
\lueckentext
Für $v \ll c$ ist $\gamma \approx$ \lsg[1.5cm]{1}. Dann liefern die Formeln der Relativitätstheorie dieselben Ergebnisse wie die\\ \lsg[5.25cm]{klassische Mechanik}. Sie ist der \lsg[3.5cm]{Grenzfall} für kleine Geschwindigkeiten.\\
Eine neue Theorie macht die alte nicht wertlos. Sie zeigt, wo deren \lsg[5.25cm]{Gültigkeitsbereich} endet, und muss\\ in diesem Bereich \lsg[5.25cm]{dieselben Ergebnisse} liefern.
\end{merke}
